```python
from typing import List, Tuple

class IntervalNode:
    """
    A node in the Interval Tree.
    Each node stores an interval, the maximum endpoint in its subtree,
    and references to its left and right children.
    """
    def __init__(self, interval: Tuple[int, int]):
        self.interval = interval
        self.max_end = interval[1]
        self.left = None
        self.right = None

class IntervalTree:
    """
    An Interval Tree data structure that supports insertion and overlap search.
    """
    def __init__(self):
        self.root = None

    def _insert(self, node: IntervalNode, start: int, end: int) -> IntervalNode:
        """
        Helper function to insert an interval into the tree.
        """
        if node is None:
            return IntervalNode((start, end))

        if start < node.interval[0]:
            node.left = self._insert(node.left, start, end)
        else:
            node.right = self._insert(node.right, start, end)

        node.max_end = max(node.max_end, end)
        return node

    def insert(self, start: int, end: int) -> None:
        """
        Inserts an interval into the tree.
        """
        self.root = self._insert(self.root, start, end)

    def _find_overlapping(self, node: IntervalNode, start: int, end: int, overlapping: List[Tuple[int, int]]) -> None:
        """
        Helper function to find overlapping intervals.
        """
        if node is None:
            return

        if node.interval[0] < end and node.interval[1] > start:
            overlapping.append(node.interval)

        if start <= node.max_end:
            self._find_overlapping(node.left, start, end, overlapping)

        if end > node.interval[0]:
            self._find_overlapping(node.right, start, end, overlapping)

    def find_overlapping(self, start: int, end: int) -> List[Tuple[int, int]]:
        """
        Finds all intervals that overlap with the given interval.
        """
        overlapping = []
        self._find_overlapping(self.root, start, end, overlapping)
        return overlapping

# Example usage:
tree = IntervalTree()
tree.insert(1, 3)
tree.insert(2, 5)
tree.insert(6, 8)
print(tree.find_overlapping(1, 4))  # Output: [(1, 3), (2, 5)]
print(tree.find_overlapping(5, 7))   # Output: [(2, 5)]
```

This implementation defines an `IntervalNode` class to represent each node in the interval tree, with attributes for the interval, the maximum endpoint in its subtree, and references to its left and right children. The `IntervalTree` class contains methods for inserting intervals and finding overlapping intervals.

The `_insert` helper function recursively inserts an interval into the tree, updating the maximum endpoint in each node as necessary. The `insert` method simply calls `_insert` with the root node and the interval to be inserted.

The `_find_overlapping` helper function recursively searches the tree for overlapping intervals, adding any found intervals to a list. The `find_overlapping` method calls `_find_overlapping` with the root node and the query interval, returning the list of overlapping intervals.

The type annotations and docstrings provide clear documentation of the classes and methods, making the code easier to understand and use. The average time complexity for overlap searches is O(log N), due to the balanced nature of the interval tree.
