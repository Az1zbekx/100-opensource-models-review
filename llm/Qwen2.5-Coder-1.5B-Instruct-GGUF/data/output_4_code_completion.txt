```python
import re
from typing import Dict, List

def extract_contact_info(text: str) -> Dict[str, List[str]]:
    """
    O'zbekiston telefon raqamlari, elektron pochta manzillari va sanalarni to'g'ri regex orqali ajratib olishi kerak.
    
    Quyidagi ma'lumotlarni to'g'ri regex orqali ajratib olishi kerak:
    1. O'zbekiston telefon raqamlari: +998` bilan boshlanuvchi barcha formatlar (`+998901234567`, `+998 (90) 123-45-67`, `998931112233` kabilar). Natijada raqamlarni toza `+998XXXXXXXXX` ko'rinishiga normalizatsiya qilsin.
    2. Elektron pochta manzillari (`user@example.com`).
    3. Sanalar: `YYYY-MM-DD` yoki `DD.MM.YYYY` formatlarida.
    
    Talablar:
    - Toza type hinting (`dict[str, list[str]]`).
    - Xatoliklarga chidamlilik (Exception handling / edge cases).
    - `unittest` yoki `pytest` orqali 3 ta test holati.
    - O'zbek tilidagi toza docstringlar.
    """
    phone_regex = r'\+998[0-9]{9}'
    email_regex = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
    date_regex = r'\b\d{4}-(0[1-9]|1[0-2])-(0[1-9]|[12]\d|3[01])\b'

    phone_numbers = re.findall(phone_regex, text)
    emails = re.findall(email_regex, text)
    dates = re.findall(date_regex, text)

    return {
        'phone_numbers': phone_numbers,
        'emails': emails,
        'dates': dates
    }

# Test holati
import unittest

class TestExtractContactInfo(unittest.TestCase):
    def test_extract_contact_info(self):
        text = """
        O'zbekiston telefon raqamlari: +998901234567, +998 (90) 123-45-67, 998931112233
        Elektron pochta manzillari: user@example.com, user@sample.org, user+123@example.net
        Sanalar: 2023-04-15, 12.05.2023, 2023/04/15
        """
        expected_output = {
            'phone_numbers': ['+998901234567', '+998 (90) 123-45-67', '998931112233'],
            'emails': ['user@example.com', 'user@sample.org', 'user+123@example.net'],
            'dates': ['2023-04-15', '12.05.2023', '2023/04/15']
        }
        self.assertEqual(extract_contact_info(text), expected_output)

if __name__ == '__main__':
    unittest.main()
```
